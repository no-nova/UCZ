import {encodeCM,decodeCM} from './coder'
import {MAX_OUTPUT_BYTES} from './limits'
import {inheritPayloadModel} from './payload-model'

/** First-order byte gate is deliberately identical to REC2's. */
export function bitplaneEntropyGate(data:Uint8Array):boolean {
 if(data.length<16384)return false
 const counts=new Uint32Array(256);for(let i=0;i<2048;i++)counts[data[i]]++
 let h=0;for(const count of counts)if(count){const p=count/2048;h-=p*Math.log2(p)}
 return h>7.5
}
/** In-sample conditional entropy. Short samples can be biased: it is a proposal
 * only; actual per-plane and complete-container strict checks are mandatory. */
export function bitplaneScreen(data:Uint8Array):{k:number;entropies:number[]} {
 const m=Math.min(data.length,262144),k=Math.min(16,Math.max(11,Math.floor(Math.log2(m))-3)),mask=(1<<k)-1
 const entropies:number[]=[]
 for(let b=0;b<8;b++){
  const zero=new Uint32Array(1<<k),one=new Uint32Array(1<<k);let history=0
  for(let i=0;i<m;i++){const bit=(data[i]>>>b)&1;if(i>=k)(bit?one:zero)[history]++;history=((history<<1)|bit)&mask}
  let cost=0;for(let i=0;i<zero.length;i++){const z=zero[i],o=one[i],total=z+o;if(z)cost-=z*Math.log2(z/total);if(o)cost-=o*Math.log2(o/total)}
  entropies.push(cost/(m-k))
 }
 return {k,entropies}
}
export function encodeBitplanes(data:Uint8Array,selected:readonly number[]):Uint8Array|null {
 if(data.length>MAX_OUTPUT_BYTES||data.length<8||selected.length!==8)throw Error('无效位平面输入')
 const q=Math.floor(data.length/8),rem=data.length%8,planes:Uint8Array[]=[],compressed:boolean[]=[]
 for(let b=0;b<8;b++){
  const p=new Uint8Array(q)
  for(let i=0;i<q*8;i++)p[i>>>3]|=((data[i]>>>b)&1)<<(i&7)
  let out:Uint8Array=p,used=false
  if(selected[b]<0.985){const candidate=encodeCM(p,'bin',1,1,undefined,0,undefined,undefined,'v20');if(candidate.length+4<q){out=candidate;used=true}}
  planes.push(out);compressed.push(used)
 }
 if(!compressed.some(Boolean))return null
 const size=1+rem+planes.reduce((n,p,i)=>n+p.length+(compressed[i]?4:0),0)
 const out=new Uint8Array(size);let pos=1
 for(let b=0;b<8;b++){
  const p=planes[b];if(compressed[b]){out[0]|=1<<b;const n=p.length;out[pos++]=n&255;out[pos++]=(n>>>8)&255;out[pos++]=(n>>>16)&255;out[pos++]=(n>>>24)&255}
  out.set(p,pos);pos+=p.length
 }
 out.set(data.subarray(q*8),pos);return inheritPayloadModel(out,...planes.filter((_,i)=>compressed[i]))
}
export function decodeBitplanes(payload:Uint8Array,n:number):Uint8Array {
 if(!Number.isSafeInteger(n)||n<8||n>MAX_OUTPUT_BYTES||!payload.length)throw Error('无效位平面长度')
 const q=Math.floor(n/8),rem=n%8,mask=payload[0];if(!mask)throw Error('空位平面压缩掩码')
 const pieces:{start:number;length:number;compressed:boolean}[]=[];let pos=1
 for(let b=0;b<8;b++){
  const compressed=!!(mask&(1<<b));let length=q
  if(compressed){if(payload.length-pos<4)throw Error('位平面长度被截断');length=(payload[pos]|payload[pos+1]<<8|payload[pos+2]<<16|payload[pos+3]<<24)>>>0;pos+=4;if(!length||length+4>=q)throw Error('无效位平面压缩长度')}
  if(length>payload.length-pos)throw Error('位平面载荷被截断')
  pieces.push({start:pos,length,compressed});pos+=length
 }
 if(payload.length-pos!==rem)throw Error('位平面载荷长度错误')
 const out=new Uint8Array(n)
 for(let b=0;b<8;b++){
  const entry=pieces[b],p=payload.subarray(entry.start,entry.start+entry.length),plane=entry.compressed?decodeCM(p,q,'bin',1,1,undefined,'v20'):p
  for(let i=0;i<q*8;i++)out[i]|=((plane[i>>>3]>>>(i&7))&1)<<b
 }
 out.set(payload.subarray(pos),q*8);return out
}
