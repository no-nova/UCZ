// P16.2: bit-level context model for bit planes.
//
// The existing bitplane path packs a plane into ceil(n/8) bytes and calls the
// byte-oriented BIN CM. That can only see a k-bit history as a fractional
// byte prefix, so an order-16..20 bit structure (AITDCC/L planes 0..3 report
// 16..20-order structure with causalH20 0.54..0.97) is invisible to it: the
// packed stream's own models are orders of *bytes*.
//
// Here the model works directly on bits with an explicit history of k bits and
// a logistic mixture of several orders. No sample-based screen is used to
// decide adoption: a plane is only stored compressed when the real coder beat
// the raw plane, so a random plane can never be adopted by mistake.
import {ByteWriter} from './byte-writer'

const PBITS = 12
const PSCALE = 1 << PBITS
const PINIT = PSCALE >> 1
const STRETCH_MAX = 2047

// stretch(p) = ln(p/(1-p)) * 256, clamped; squash is its inverse.
const stretchTable = new Int16Array(PSCALE)
for (let p = 0; p < PSCALE; p++) {
  const x = Math.max(1 / PSCALE, Math.min(1 - 1 / PSCALE, (p + 0.5) / PSCALE))
  stretchTable[p] = Math.max(-STRETCH_MAX, Math.min(STRETCH_MAX, Math.round(Math.log(x / (1 - x)) * 256)))
}
const squashTable = new Int16Array(2 * STRETCH_MAX + 1)
for (let i = 0; i <= 2 * STRETCH_MAX; i++) {
  const x = (i - STRETCH_MAX) / 256
  squashTable[i] = Math.max(1, Math.min(PSCALE - 1, Math.round(PSCALE / (1 + Math.exp(-x)))))
}
// mixed is fractional; the table index must be an integer or the lookup
// silently yields undefined (NaN probabilities, a 5-byte "encoder").
const squash = (st: number) => squashTable[Math.max(-STRETCH_MAX, Math.min(STRETCH_MAX, Math.round(st))) + STRETCH_MAX]

/** Counts are n0<<16|n1 and are halved before they can overflow the mixer. */
const MAX_COUNT = 1024

export interface BitCMModel {
  orders: number[]
  masks: number[]
  counts: Uint32Array[]
  weights: Float64Array
}

export function createBitCMModel(orders: readonly number[]): BitCMModel {
  if (!orders.length || orders.some(o => !Number.isInteger(o) || o < 1 || o > 24)) throw Error('无效位历史阶数')
  return {
    orders: [...orders],
    masks: orders.map(o => (1 << o) - 1),
    counts: orders.map(o => new Uint32Array(1 << o)),
    weights: initialWeights(orders.length),
  }
}

function probOf(count: number): number {
  const n0 = count >>> 16, n1 = count & 0xffff
  const total = 2 * (n0 + n1) + 2
  return Math.max(1, Math.min(PSCALE - 1, Math.floor(((2 * n1 + 1) * PSCALE) / total)))
}

function updateCount(counts: Uint32Array, index: number, bit: number): number {
  const count = counts[index]
  let n0 = count >>> 16, n1 = count & 0xffff
  if (bit) n1++; else n0++
  const next = n0 + n1 > MAX_COUNT ? ((n0 + 1) >> 1) << 16 | ((n1 + 1) >> 1) : (n0 << 16) | n1
  counts[index] = next
  return index
}

interface MixState { p: number; stretch: number[] }

// Mixture weights are unnormalised and clamped; a constant bias input lets the
// mixer express "no information" (p=0.5) without depending on the model weights.
const BIAS_STRETCH = 256
const INIT_WEIGHT = 0.35
const LR = 0.0025

function initialWeights(n: number): Float64Array {
  const w = new Float64Array(n + 1)
  for (let i = 0; i < n; i++) w[i] = INIT_WEIGHT
  return w
}

/** One shared model step for encode and decode: identical order and updates. */
function modelPredict(m: BitCMModel, history: number, st: number[]): MixState {
  let mixed = 0
  for (let i = 0; i < m.orders.length; i++) {
    const prob = probOf(m.counts[i][history & m.masks[i]])
    st[i] = stretchTable[prob]
    mixed += m.weights[i] * st[i]
  }
  mixed += m.weights[m.orders.length] * BIAS_STRETCH
  return { p: squash(mixed), stretch: st }
}

function modelUpdate(m: BitCMModel, predict: MixState, bit: number, history: number, lr: number): void {
  const err = bit * PSCALE - predict.p
  const scale = lr / (PSCALE * STRETCH_MAX)
  for (let i = 0; i < m.orders.length; i++) {
    const next = m.weights[i] + err * predict.stretch[i] * scale
    m.weights[i] = next < 0 ? 0 : next > 4 ? 4 : next
    updateCount(m.counts[i], history & m.masks[i], bit)
  }
  const bias = m.weights[m.orders.length] + err * BIAS_STRETCH * scale
  m.weights[m.orders.length] = bias < -4 ? -4 : bias > 4 ? 4 : bias
}

class RangeEncoder {
  // LZMA-style carry handling: `low` must keep its byte 3..4 bits so that the
  // carry test (low >= 2^32) still fires. Never mask low before shiftLow.
  private low = 0
  private range = 0xffffffff
  private cache = 0
  private cacheSize = 1
  private readonly out = new ByteWriter()

  /** `prob` is P(bit=1), matching the model. The lower sub-range is the 1 branch. */
  bit(prob: number, bit: number): void {
    const bound = (this.range >>> PBITS) * prob
    if (bit) this.range = bound
    else { this.low += bound; this.range -= bound }
    while (this.range < 0x01000000) { this.range = (this.range << 8) >>> 0; this.shiftLow() }
  }
  private shiftLow(): void {
    if (this.low < 0xff000000 || this.low >= 0x100000000) {
      const carry = this.low >= 0x100000000 ? 1 : 0
      let temp = this.cache
      do {
        this.out.push((temp + carry) & 0xff)
        temp = 0xff
      } while (--this.cacheSize !== 0)
      this.cache = (this.low >>> 24) & 0xff
    }
    this.cacheSize++
    this.low = (this.low & 0xffffff) * 256
  }
  finish(): Uint8Array {
    for (let i = 0; i < 5; i++) this.shiftLow()
    return this.out.finish()
  }
}

class RangeDecoder {
  private range = 0xffffffff
  private code = 0
  private pos = 0
  constructor(private readonly input: Uint8Array) {
    for (let i = 0; i < 5; i++) this.code = ((this.code << 8) | this.byte()) >>> 0
  }
  private byte(): number { return this.pos < this.input.length ? this.input[this.pos++] : 0 }
  bit(prob: number): number {
    const bound = (this.range >>> PBITS) * prob
    let bit: number
    if (this.code < bound) { this.range = bound; bit = 1 }
    else { this.code = (this.code - bound) >>> 0; this.range -= bound; bit = 0 }
    while (this.range < 0x01000000) { this.range = (this.range << 8) >>> 0; this.code = ((this.code << 8) | this.byte()) >>> 0 }
    return bit
  }
}

export function bitCMBitCount(bytes: number): number { return Math.floor(bytes) * 8 }

 const m=createBitCMModel(orders),st=new Array<number>(m.orders.length).fill(0)
 return modelPredict(m,0,st).p

/** Diagnostic only: exact model cross-entropy of the same loop the encoder runs. */
export function debugBitCMLoss(plane: Uint8Array, bits: number, orders: readonly number[], rate = LR): number {
  const model = createBitCMModel(orders), st = new Array<number>(model.orders.length).fill(0)
  let history = 0, loss = 0, minP = PSCALE, maxP = 0
  for (let i = 0; i < bits; i++) {
    const bit = (plane[i >>> 3] >>> (i & 7)) & 1
    const predict = modelPredict(model, history, st)
    if (predict.p < minP) minP = predict.p
    if (predict.p > maxP) maxP = predict.p
    loss -= Math.log2(bit ? predict.p / PSCALE : 1 - predict.p / PSCALE)
    modelUpdate(model, predict, bit, history, rate)
    history = ((history << 1) | bit) & model.masks[model.orders.length - 1]
  }
  return loss
}

/** Diagnostic only: the exact probability sequence the encoder would use. */
export function debugBitCMProbs(plane: Uint8Array, bits: number, orders: readonly number[], rate = LR): Float64Array {
  const model = createBitCMModel(orders), st = new Array<number>(model.orders.length).fill(0)
  const out = new Float64Array(bits)
  let history = 0
  for (let i = 0; i < bits; i++) {
    const bit = (plane[i >>> 3] >>> (i & 7)) & 1
    const predict = modelPredict(model, history, st)
    out[i] = predict.p
    modelUpdate(model, predict, bit, history, rate)
    history = ((history << 1) | bit) & model.masks[model.orders.length - 1]
  }
  return out
}

/** Diagnostic only: encode with a frozen probability sequence. */
export function debugEncodeFixed(plane: Uint8Array, bits: number, probs: Float64Array): Uint8Array {
  const encoder = new RangeEncoder()
  for (let i = 0; i < bits; i++) {
    const bit = (plane[i >>> 3] >>> (i & 7)) & 1
    encoder.bit(probs[i], bit)
  }
  return encoder.finish()
}

/** Encode `bits` bits read LSB-first from packed `plane`. */
export function encodePlaneBitCM(plane: Uint8Array, bits: number, model: BitCMModel, rate = LR): Uint8Array {
  const encoder = new RangeEncoder(), st = new Array<number>(model.orders.length).fill(0)
  let history = 0
  for (let i = 0; i < bits; i++) {
    const bit = (plane[i >>> 3] >>> (i & 7)) & 1
    const predict = modelPredict(model, history, st)
    encoder.bit(predict.p, bit)
    modelUpdate(model, predict, bit, history, rate)
    history = ((history << 1) | bit) & model.masks[model.orders.length - 1]
  }
  return encoder.finish()
}

/** Decode `bits` bits into a new packed plane. Symmetric with the encoder. */
export function decodePlaneBitCM(payload: Uint8Array, bits: number, model: BitCMModel, rate = LR): Uint8Array {
  const decoder = new RangeDecoder(payload), plane = new Uint8Array(Math.ceil(bits / 8))
  const st = new Array<number>(model.orders.length).fill(0)
  let history = 0
  for (let i = 0; i < bits; i++) {
    const predict = modelPredict(model, history, st)
    const bit = decoder.bit(predict.p)
    if (bit) plane[i >>> 3] |= 1 << (i & 7)
    modelUpdate(model, predict, bit, history, rate)
    history = ((history << 1) | bit) & model.masks[model.orders.length - 1]
  }
  return plane
}
