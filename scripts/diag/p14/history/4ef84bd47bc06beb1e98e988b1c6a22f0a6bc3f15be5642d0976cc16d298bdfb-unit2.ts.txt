// Unit 2: fixed-layout residual/plane experiments, using real v20 CM.
import assert from 'node:assert/strict'
import { readFileSync,writeFileSync,mkdirSync,existsSync } from 'node:fs'
import { encodeCM,decodeCM,withCMBlockSize,withCMProbe,a2Suppress,transposeInnerSuppress,p14 } from '@lib/coder'
import { wide16Fwd,wide16Inv,wordRowDelta } from '@lib/transforms'
import { planColumnGroups,gatherGroup,groupDelta,encodeSpec,decodeSpec,ID } from '@lib/codec'
import { load,eq,sha,concat,varint,done,entropy } from './common'
const [name,variant]=process.argv.slice(2),{entry,data}=load(name),begun=performance.now()
const calls:any[]=[];let encodeMs=0,decodeMs=0
function enc(x:Uint8Array,p:any,s=1,r=1,g=0) {const t=performance.now(),out=withCMBlockSize(data.length,()=>encodeCM(x,p,s,r,undefined,0,undefined,undefined,'v20',g));encodeMs+=performance.now()-t;calls.push({inputBytes:x.length,profile:p,s,r,geomStride:g,sideBytes:p14.side?.length??0,payloadBytes:out.length});return out}
function dec(x:Uint8Array,n:number,p:any,s=1,r=1,g=0) {const t=performance.now(),out=withCMBlockSize(data.length,()=>decodeCM(x,n,p,s,r,undefined,'v20',g));decodeMs+=performance.now()-t;return out}
function result(payload:Uint8Array,extra:any) {done({unit:2,input:entry,variant,kernel:'v20',...extra,payloadBytes:payload.length,payloadSha256:sha(payload),roundtrip:true,cmCalls:calls,
 encodeMs:Math.round(encodeMs),decodeMs:Math.round(decodeMs),totalWorkMs:Math.round(performance.now()-begun),scope:'fixed 256KiB-class window; actual experimental CM + framing; no full adaptive chooser or complete UCZ'})}
if(variant==='plane-info') {
 const row=entry.file==='mr'?1024:3800,mode=entry.file==='mr'?6:7,residual=wide16Fwd(data,row,mode)
 const half=data.length/2,high=residual.subarray(0,half),low=residual.subarray(half),hist=new Uint32Array(65536),hc=new Uint32Array(256)
 for(let i=0;i<half;i++){hc[high[i]]++;hist[(high[i]<<8)|low[i]]++}
 let conditional=0;for(let i=0;i<hist.length;i++)if(hist[i])conditional-=hist[i]/half*Math.log2(hist[i]/hc[i>>>8])
 done({unit:2,input:entry,variant,analysisOnly:true,cmCalls:0,rowBytes:row,mode,planeBytes:half,
  highZeroFraction:hc[0]/half,highEntropy:entropy(high),lowEntropy:entropy(low),empiricalLowGivenHighEntropy:conditional,empiricalMutualInformation:entropy(low)-conditional,
  warning:'plug-in same-sample entropy, not CM cost or a generalization guarantee; MSB is ZigZag residual, not original image pixel',scope:'transformed 256KiB window only'})
}else if(!variant.startsWith('tg-')) {
 assert.ok(entry.file==='mr'||entry.file==='x-ray')
 const row=entry.file==='mr'?1024:3800,mode=variant.startsWith('wide3')?3:entry.file==='mr'?6:7
 const transformed=wide16Fwd(data,row,mode);eq(wide16Inv(transformed,row,mode),data)
 a2Suppress.depth++
 try {
  let payload:Uint8Array,reconstructed:Uint8Array
  if(mode<6) {
   const g=variant==='wide3-geom'?row:0
   payload=enc(transformed,'res2d',2,row,g);reconstructed=dec(payload,data.length,'res2d',2,row,g)
  }else {
   const half=data.length/2,high=transformed.subarray(0,half),low=transformed.subarray(half)
   const hp=variant.startsWith('high2d')?'res2d':'res',hr=hp==='res2d'?row/2:1
   const h=enc(high,hp,1,hr),hd=dec(h,half,hp,1,hr);eq(hd,high)
   const side=['msb','msb-add','msb-detail','high2d-msb'].includes(variant)
   if(side)p14.side=high
   const g=variant==='geom'?row/2:0,l=enc(low,'res2d',1,row/2,g)
   if(side)p14.side=hd // Decoder uses ONLY the already-decoded first stream.
   const ld=dec(l,half,'res2d',1,row/2,g);p14.side=undefined
   payload=concat(varint(h.length),h,l);reconstructed=concat(hd,ld)
  }
  eq(reconstructed,transformed);eq(wide16Inv(reconstructed,row,mode),data)
  if(variant==='control'&&name==='mr-mid')assert.equal(payload.length,49237,'must reproduce exact production P13 residual oracle')
  if(variant==='control'&&name==='x-ray-mid')assert.equal(payload.length,115212,'must reproduce exact production P13 residual oracle')
  result(payload,{spec:{id:31,s:mode,r:row},productionCompatible:['control','wide3'].includes(variant),residualSha256:sha(transformed),sideIsAlreadyDecodedPlane:true})
 }finally{a2Suppress.depth--;p14.side=undefined}
}else {
 assert.equal(entry.file,'sao')
 const width=28,planPath=`.diag-state/p14-plans/${name}.json`
 if(variant==='tg-plan') {
  assert.ok(!existsSync(planPath),'immutable prepared plan already exists')
  const sample=data.subarray(0,Math.floor(Math.min(65536,data.length)/width)*width)
  const groups=withCMProbe(()=>planColumnGroups([sample],width,true,false))
  const record={input:entry,kind:'fixed-plan-preparation-not-full-payload',sourceCodecSha256:sha(readFileSync('src/lib/codec.ts')),planningInputBytes:sample.length,
   plannerInternalWindowMaxBytes:16000,planningKernel:'production v13 (as current TG3 planner)',groups,durationMs:Math.round(performance.now()-begun)}
  mkdirSync('.diag-state/p14-plans',{recursive:true});writeFileSync(planPath,JSON.stringify(record,null,2)+'\n',{flag:'wx'})
  done(record)
 }else if(variant==='tg-flat'||variant==='tg-flat-side') {
  // New candidate layout, NOT the production TG3 layout. Exactly width byte planes.
  assert.equal(data.length%width,0)
  const planeLen=data.length/width,flat=new Uint8Array(data.length)
  for(let c=0;c<width;c++)for(let r=0;r<planeLen;r++)flat[c*planeLen+r]=data[r*width+c]
  transposeInnerSuppress.depth++
  try {
   if(variant==='tg-flat-side')p14.planeLen=planeLen
   const payload=enc(flat,'bin'),out=dec(payload,flat.length,'bin');eq(out,flat)
   const raw=new Uint8Array(data.length);for(let c=0;c<width;c++)for(let r=0;r<planeLen;r++)raw[r*width+c]=out[c*planeLen+r]
   eq(raw,data);result(payload,{layout:'new 28 single-byte planes in one CM stream; not TG3 grouped layout',planeLen,productionCompatible:false,framingBytesExcluded:'new layout marker not specified; no deployable format',flatSha256:sha(flat)})
  }finally{transposeInnerSuppress.depth--;p14.planeLen=0}
 }else {
  const planBytes=readFileSync(planPath),plan=JSON.parse(planBytes.toString())
  assert.equal(plan.input.sha256,entry.sha256);assert.equal(plan.sourceCodecSha256,sha(readFileSync('src/lib/codec.ts')))
  const outputs:Uint8Array[]=[],reconstructed=new Uint8Array(data.length),channels:any[]=[]
  let prevEncoded:{data:Uint8Array,span:number}|undefined,prevDecoded:{data:Uint8Array,span:number}|undefined
  const sideFor=(n:number,span:number,prev:{data:Uint8Array,span:number}|undefined)=>{
   if(!prev)return undefined
   const side=new Uint8Array(n)
   for(let i=0;i<n;i++)side[i]=prev.data[Math.floor(i/span)*prev.span+Math.min(i%span,prev.span-1)]??0
   return side
  }
  transposeInnerSuppress.depth++
  try {
   for(const group of plan.groups) {
    const {data:x,lengths}=gatherGroup([data],width,group.start,group.span,group.reverse)
    let mode=group.mode;if(mode===ID.FILL&&x.some((v:number)=>v!==x[0]))mode=ID.BIN
    const wd=mode===7||mode===8,profile=mode===ID.O0?'o0':mode===ID.DELTA?'res':'bin'
    const s=mode===ID.O0?1:group.span,g=(mode===ID.BIN||wd)?group.span:0
    const transformed=wd?wordRowDelta(x,lengths,group.span,false,!!group.reverse):mode===ID.DELTA?groupDelta(x,lengths,group.span,false):x
    let payload:Uint8Array,out:Uint8Array
    const isCM=[ID.O0,ID.BIN,ID.DELTA,7,8].includes(mode)
    if(variant==='tg-side')p14.side=sideFor(x.length,group.span,prevEncoded)
    if(isCM)payload=enc(transformed,profile,s,1,g)
    else payload=mode===ID.STORE?x:encodeSpec(x,{id:mode,s:1,r:1})
    if(mode!==ID.STORE&&payload.length>=x.length){mode=ID.STORE;payload=x}
    if(variant==='tg-side')p14.side=sideFor(x.length,group.span,prevDecoded)
    if(mode===ID.STORE)out=payload.slice()
    else if(isCM) {
     const d=dec(payload,x.length,profile,s,1,g)
     out=wd?wordRowDelta(d,lengths,group.span,true,!!group.reverse):mode===ID.DELTA?groupDelta(d,lengths,group.span,true):d
    }else out=decodeSpec(payload,x.length,{id:mode,s:1,r:1})
    eq(out,x)
    outputs.push(concat(Uint8Array.of(group.span,mode|(group.reverse?128:0)),varint(payload.length),payload))
    let k=0
    for(let row=0;row<data.length;row+=width) {
     const a=Math.min(data.length,row+group.start),b=Math.min(data.length,a+group.span)
     if(group.reverse&&b-a===group.span)for(let i=b-1;i>=a;i--)reconstructed[i]=out[k++]
     else for(let i=a;i<b;i++)reconstructed[i]=out[k++]
    }
    channels.push({...group,actualMode:mode,rawBytes:x.length,payloadBytes:payload.length})
    prevEncoded={data:x,span:group.span};prevDecoded={data:out,span:group.span}
   }
   eq(reconstructed,data)
   result(concat(...outputs),{layout:'actual TG3 gather/groupDelta/wordRowDelta/CM, fixed plan proposed on <=64KiB; no full 256KiB plan search',planSha256:sha(planBytes),channels,productionCompatible:variant==='tg-control',sideIsAlreadyDecodedPreviousGroup:true})
  }finally{transposeInnerSuppress.depth--;p14.side=undefined}
 }
}
